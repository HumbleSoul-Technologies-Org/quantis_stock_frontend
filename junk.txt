const openDB = indexedDB.open('erp-offline-db');

openDB.onsuccess = (event) => {
  const db = event.target.result;
  const tx = db.transaction('operations', 'readwrite');
  const store = tx.objectStore('operations');
  
  // Get all operations
  const getAllOps = store.getAll();
  
  getAllOps.onsuccess = () => {
    const ops = getAllOps.result;
    const failedOp = ops.find(o => o.status === 'failed');
    
    if (!failedOp) {
      console.log('❌ No failed operations found');
      return;
    }
    
    console.log('📋 Found failed operation:', {
      id: failedOp.id,
      type: failedOp.type,
      status: failedOp.status,
      retries: failedOp.retries
    });
    
    // Reset to pending with 0 retries
    const updated = {
      ...failedOp,
      status: 'pending',
      retries: 0,
      updatedAt: Date.now()
    };
    
    const updateReq = store.put(updated);
    
    updateReq.onsuccess = () => {
      console.log('✅ Operation reset to pending!', {
        id: updated.id,
        type: updated.type,
        newStatus: updated.status,
        newRetries: updated.retries,
        newUpdatedAt: new Date(updated.updatedAt).toISOString()
      });
      console.log('🔄 Refresh the page and the supplier should now sync!');
    };
    
    updateReq.onerror = () => {
      console.error('❌ Failed to update operation:', updateReq.error);
    };
  };
  
  getAllOps.onerror = () => {
    console.error('❌ Failed to fetch operations:', getAllOps.error);
  };
};

openDB.onerror = () => {
  console.error('❌ Failed to open IndexedDB');
};
